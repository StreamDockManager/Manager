cmake_minimum_required(VERSION 3.25)
project(sdm_hyprland_keyboard_guard VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This optional compositor module must be built against the target Linux compositor SDK")
endif()
set(SDM_MANAGER_EXECUTABLE "" CACHE FILEPATH "Exact installed Manager executable allowed to request scoped keyboard observation")
option(SDM_CROSS_BUILD_TARGET_PATH "Use an explicitly supplied canonical target path, not an executable installed in the builder" OFF)
if(NOT IS_ABSOLUTE "${SDM_MANAGER_EXECUTABLE}")
  message(FATAL_ERROR "Set SDM_MANAGER_EXECUTABLE to the absolute installed Manager executable path")
endif()
if(SDM_CROSS_BUILD_TARGET_PATH)
  cmake_path(NORMAL_PATH SDM_MANAGER_EXECUTABLE OUTPUT_VARIABLE SDM_NORMALIZED_PATH)
  if(NOT SDM_NORMALIZED_PATH STREQUAL SDM_MANAGER_EXECUTABLE)
    message(FATAL_ERROR "The cross-build target path must be normalized and canonical on the target machine")
  endif()
  message(STATUS "Explicit cross-build target allowlist: ${SDM_MANAGER_EXECUTABLE}; verify target realpath before loading")
elseif(EXISTS "${SDM_MANAGER_EXECUTABLE}" AND NOT IS_DIRECTORY "${SDM_MANAGER_EXECUTABLE}")
  file(REAL_PATH "${SDM_MANAGER_EXECUTABLE}" SDM_MANAGER_EXECUTABLE)
else()
  message(FATAL_ERROR "Set SDM_MANAGER_EXECUTABLE to an existing executable, or explicitly enable SDM_CROSS_BUILD_TARGET_PATH for a target-only path")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(HYPRLAND REQUIRED IMPORTED_TARGET hyprland=0.56.2)
pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-server)
find_package(nlohmann_json 3.11 REQUIRED)
add_library(sdm-keyboard-guard MODULE main.cpp)
target_compile_features(sdm-keyboard-guard PRIVATE cxx_std_23)
target_link_libraries(sdm-keyboard-guard PRIVATE PkgConfig::HYPRLAND PkgConfig::WAYLAND nlohmann_json::nlohmann_json)
target_compile_definitions(sdm-keyboard-guard PRIVATE SDM_MANAGER_EXECUTABLE="${SDM_MANAGER_EXECUTABLE}")
set_target_properties(sdm-keyboard-guard PROPERTIES PREFIX "")
# Deliberately no install rule, compositor configuration changes, or load command
